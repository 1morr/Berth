## F. Handing 100999 files back without sudo (05:13, ~/berth-trial-42/chown-test)

$ docker run --rm -v "$PWD":/d alpine sh -c "mkdir -p /d/a/b && touch /d/a/b/f && chown -R 1000:1000 /d/a"
$ ls -lnR a
a:
drwxr-xr-x 2 100999 100999 4096 Oct  9 05:13 b
$ rm -rf a
rm: cannot remove 'a/b/f': Permission denied
$ docker run --rm -v "$PWD":/d alpine chown -R 0:0 /d/a
$ ls -lnR a
a:
drwxr-xr-x 2 1000 1000 4096 Oct  9 05:13 b
$ rm -rf a && echo removed-ok
removed-ok
